from flask import Flask, render_template, request, jsonify
from flask_cors import CORS
import json
import uuid
import base64
import urllib.parse

app = Flask(__name__)
CORS(app)

# صفحه اصلی
@app.route('/')
def index():
    return render_template('index.html')

# API برای ساخت کانفیگ (اختیاری - برای استفاده در فرانت‌اند)
@app.route('/api/generate', methods=['POST'])
def generate_config():
    data = request.json
    
    server = data.get('server')
    uuid_val = data.get('uuid', str(uuid.uuid4()))
    ws_path = data.get('path', '/v2ray')
    port = data.get('port', 443)
    tls = data.get('tls', True)
    name = data.get('name', 'My V2Ray')
    
    if not server:
        return jsonify({'error': 'آدرس سرور الزامی است'}), 400
    
    config = {
        'v': '2',
        'ps': name,
        'add': server,
        'port': int(port),
        'id': uuid_val,
        'aid': '0',
        'net': 'ws',
        'type': 'none',
        'host': '',
        'path': ws_path,
        'tls': 'tls' if tls else '',
        'sni': '',
        'alpn': '',
        'fp': 'chrome'
    }
    
    vmess_json = json.dumps(config)
    vmess_base64 = base64.b64encode(vmess_json.encode()).decode()
    vmess_link = f"vmess://{vmess_base64}"
    
    return jsonify({
        'link': vmess_link,
        'config': config,
        'json': json.dumps(config, indent=2)
    })

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000, debug=True)
